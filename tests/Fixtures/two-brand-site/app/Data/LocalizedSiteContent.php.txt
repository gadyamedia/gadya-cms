<?php

namespace App\Data;

use Illuminate\Support\Facades\Lang;

/**
 * Serves the site's structured content in the active locale.
 *
 * English lives in {@see SiteContent}. Translations live in `lang/{locale}/` keyed by
 * slug and override only the fields they define.
 */
class LocalizedSiteContent
{
    /** @return array<int, array<string, mixed>> */
    public static function services(?string $locale = null): array
    {
        return self::translate(SiteContent::services(), 'services', $locale);
    }

    /** @return array<int, array<string, mixed>> */
    public static function locations(?string $locale = null): array
    {
        return self::translate(SiteContent::locations(), 'locations', $locale);
    }

    /**
     * @param  array<int, array<string, mixed>>  $items
     * @return array<int, array<string, mixed>>
     */
    private static function translate(array $items, string $group, ?string $locale): array
    {
        $locale ??= app()->getLocale();

        if ($locale === 'en' || ! Lang::has($group, $locale, false)) {
            return $items;
        }

        $overrides = Lang::get($group, [], $locale, false);

        if (! is_array($overrides)) {
            return $items;
        }

        return array_map(
            fn (array $item): array => array_replace($item, $overrides[$item['slug']] ?? []),
            $items,
        );
    }
}
